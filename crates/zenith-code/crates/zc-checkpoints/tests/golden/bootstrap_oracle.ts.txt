
// ---- zc golden: appended by crates/zenith-code/crates/zc-checkpoints/tests/bootstrap.rs to a
// scratch copy of server.test.ts (it reuses that file's `buildAppUnderTest`). Each scenario
// dispatches a bootstrap `thread.turn.start` over the real WebSocket RPC route with the
// orchestration engine, git driver and setup-script runner mocked, and writes
// `${ZC_GOLDEN_OUT}/<scenario>.json`: the encoded commands the engine received and the result.
import * as ZcSchema from "effect/Schema";
import * as ZcNodeFS from "node:fs";
import { OrchestrationCommand as ZcOrchestrationCommand } from "@t3tools/contracts";

const zcEncodeCommand = ZcSchema.encodeSync(ZcSchema.toCodecJson(ZcOrchestrationCommand));

type ZcScenario = {
  readonly name: string;
  readonly isRepository: boolean;
  readonly hasCommit: boolean;
  readonly createWorktree: "ok" | "defect";
  readonly prepareWorktree: boolean;
  readonly runSetupScript: boolean;
  readonly branch: string | null;
  readonly setup: "none" | "fails" | "started";
  readonly failSetupActivityAttempt?: number;
  readonly failThreadDelete?: boolean;
};

const zcScenarios: ReadonlyArray<ZcScenario> = [
  { name: "non-repo", isRepository: false, hasCommit: true, createWorktree: "defect", prepareWorktree: true, runSetupScript: true, branch: null, setup: "none" },
  { name: "no-commit", isRepository: true, hasCommit: false, createWorktree: "defect", prepareWorktree: true, runSetupScript: true, branch: "main", setup: "none" },
  { name: "setup-failure", isRepository: true, hasCommit: true, createWorktree: "ok", prepareWorktree: true, runSetupScript: true, branch: "main", setup: "fails" },
  { name: "setup-activity-failure", isRepository: true, hasCommit: true, createWorktree: "ok", prepareWorktree: true, runSetupScript: true, branch: "main", setup: "started", failSetupActivityAttempt: 2 },
  { name: "worktree-defect", isRepository: true, hasCommit: true, createWorktree: "defect", prepareWorktree: true, runSetupScript: false, branch: "main", setup: "none" },
  { name: "cleanup-failure", isRepository: true, hasCommit: true, createWorktree: "defect", prepareWorktree: true, runSetupScript: false, branch: "main", setup: "none", failThreadDelete: true },
  { name: "create-only", isRepository: true, hasCommit: true, createWorktree: "ok", prepareWorktree: false, runSetupScript: false, branch: null, setup: "none" },
];

it.layer(NodeServices.layer)("zc golden bootstrap", (it) => {
  for (const scenario of zcScenarios) {
    it.effect(`zc golden ${scenario.name}`, () =>
      Effect.gen(function* () {
        const dispatched: Array<OrchestrationCommand> = [];
        let setupActivityAttempt = 0;
        yield* buildAppUnderTest({
          layers: {
            ...(scenario.isRepository ? { vcsDriver: { isInsideWorkTree: () => Effect.succeed(true) } } : {}),
            gitVcsDriver: {
              execute: () =>
                Effect.succeed(
                  scenario.hasCommit
                    ? SUCCESSFUL_GIT_EXECUTION
                    : { ...SUCCESSFUL_GIT_EXECUTION, exitCode: ChildProcessSpawner.ExitCode(128), stderr: "fatal: Needed a single revision" },
                ),
              createWorktree: () =>
                scenario.createWorktree === "ok"
                  ? Effect.succeed({ worktree: { refName: "t3code/bootstrap-refName", path: "/tmp/bootstrap-worktree" } })
                  : Effect.die(new Error("worktree exploded")),
            },
            orchestrationEngine: {
              dispatch: (command) => {
                if (command.type === "thread.activity.append" && command.activity.kind.startsWith("setup-script.")) {
                  setupActivityAttempt += 1;
                  if (setupActivityAttempt === scenario.failSetupActivityAttempt) {
                    return Effect.fail(
                      new PersistenceSqlError({ operation: "OrchestrationEventStore.append:query", detail: "failed to append setup-script.started activity" }),
                    );
                  }
                }
                dispatched.push(command);
                if (scenario.failThreadDelete && command.type === "thread.delete") {
                  return Effect.fail(new PersistenceSqlError({ operation: "OrchestrationEventStore.append:query", detail: "thread cleanup exploded" }));
                }
                return Effect.succeed({ sequence: dispatched.length });
              },
              readEvents: () => Stream.empty,
            },
            projectSetupScriptRunner: {
              runForThread: (input) =>
                scenario.setup === "fails"
                  ? Effect.fail(
                      new ProjectSetupScriptRunner.ProjectSetupScriptOperationError({
                        threadId: input.threadId,
                        worktreePath: input.worktreePath,
                        operation: "openTerminal",
                        cause: { message: "pty unavailable" },
                      }),
                    )
                  : scenario.setup === "started"
                    ? Effect.succeed({
                        status: "started" as const,
                        scriptId: "setup",
                        scriptName: "Setup",
                        scriptCommand: "npm install",
                        terminalId: "setup-setup",
                        cwd: "/tmp/bootstrap-worktree",
                        async: true,
                      })
                    : Effect.succeed({ status: "no-script" as const }),
            },
          },
        });
        const createdAt = "2026-01-01T00:00:00.000Z";
        const threadId = ThreadId.make(`thread-golden-${scenario.name}`);
        const wsUrl = yield* getWsServerUrl("/ws");
        const result = yield* Effect.scoped(
          withWsRpcClient(wsUrl, (client) =>
            client[ORCHESTRATION_WS_METHODS.dispatchCommand]({
              type: "thread.turn.start",
              commandId: CommandId.make(`cmd-golden-${scenario.name}`),
              threadId,
              message: { messageId: MessageId.make(`msg-golden-${scenario.name}`), role: "user", text: "hello", attachments: [] },
              modelSelection: defaultModelSelection,
              runtimeMode: "full-access",
              interactionMode: "default",
              bootstrap: {
                createThread: {
                  projectId: defaultProjectId,
                  title: "Bootstrap Thread",
                  modelSelection: defaultModelSelection,
                  runtimeMode: "full-access",
                  interactionMode: "default",
                  branch: scenario.branch,
                  worktreePath: null,
                  createdAt,
                },
                ...(scenario.prepareWorktree
                  ? { prepareWorktree: { projectCwd: "/tmp/project", baseBranch: "main", branch: "t3code/bootstrap-refName" } }
                  : {}),
                runSetupScript: scenario.runSetupScript,
              },
              createdAt,
            }),
          ).pipe(Effect.result),
        );
        const out = process.env.ZC_GOLDEN_OUT;
        if (out) {
          ZcNodeFS.writeFileSync(
            `${out}/${scenario.name}.json`,
            JSON.stringify({
              commands: dispatched.map((command) => zcEncodeCommand(command)),
              result:
                result._tag === "Success"
                  ? { ok: result.success }
                  : {
                      error: {
                        message: (result.failure as { message?: string }).message,
                        bootstrapThreadDisposition: (result.failure as { bootstrapThreadDisposition?: string }).bootstrapThreadDisposition ?? null,
                      },
                    },
            }),
          );
        }
      }).pipe(Effect.provide(NodeHttpServer.layerTest)),
    );
  }
});
